export const useSettingsStore = create<SettingsStore>()(
  persist(
    (set) => ({
      ...DEFAULT_SETTINGS,
 
      setLanguage: (language: string) => {
        set({ language })
        i18n.changeLanguage(language)
        localStorage.setItem('settings:language', language)
      },
 
      setTheme: (theme: ThemeMode) => {
